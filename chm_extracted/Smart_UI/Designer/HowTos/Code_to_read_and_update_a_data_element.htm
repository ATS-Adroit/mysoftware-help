Code to read and update a data element

Add the following code to your script to read and update a data element, which in this case is called "a2.de":

//These declarations are fully qualified (could be put into the using statement:

VIZNET.Shared.DataElements.Collections.DataElementCollection __dec =

  VIZNET.Shared.DataElements.Engine.DataElementEngine.NewDataElementCollection();

       

//Create a DE with a value

DataElement __de = DataElementEngine.NewDataElement("a2.de");

__de.Value = 1223;

//Add the DE to the DE Collection (DEC)

__dec.Add(__de);

MethodReturnInfo __mri = MyConnection.UpdateDataElements(__dec);

//Check that the write was successful:

if (!__mri.Success)

{

      MyApplication.ShowInformation("Write Failed: ",  __mri.ReturnInformation + " " + __de.Name);

}

//_____________________________________________________________________

 

//Do a read by name:

__mri = MyConnection.ReadDataElements(new string[] { __de.Name });

 

//Was the read successful?   

if (__mri.Success == true )

{

        //Get the return collection of dataelements:

      DataElementCollection __dec2 = __mri.ReturnObject as DataElementCollection;

 

      if(__dec2 != null)

      {

            //Get all the return DE names:

            string [] __retNames = __dec2.ReturnAllElementNames();

            for (int i = 0;i<__retNames.Length; i++)

            {

                        MessageBox.Show(__retNames[i]);

            }                

                         

            //Get the value from the DEC by name and show the value:

            MyApplication.ShowInformation("Read Successful: " + __de.Name, 

                   __dec2.GetElementByKeyValue( __de.Name).Value.ToString());

      }

}
